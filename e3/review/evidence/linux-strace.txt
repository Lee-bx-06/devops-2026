-- 编译与链接（execve）；注意编译器只读 main.c 与 config.h --
execve("/usr/bin/cc", ["cc", "-O0", "-c", "main.c", "-o", "main.o"], ...) = 0
openat(AT_FDCWD, "main.c",  O_RDONLY|O_NOCTTY) = 3
openat(AT_FDCWD, "config.h", O_RDONLY|O_NOCTTY) = 4
openat(AT_FDCWD, "main.o", O_RDWR|O_CREAT|O_TRUNC, 0666) = 3
execve("/usr/bin/cc", ["cc", "-O0", "-o", "app", "main.o"], ...) = 0
openat(AT_FDCWD, "app", O_RDWR|O_CREAT|O_TRUNC, 0666) = 3

（newfstatat("config.h", ...) 在本次 make 中不存在）

-- unused.h：被 make stat（声明的前置依赖），但编译器从不打开它（RD）--
